       ```ts
       const connectionString = process.env.DATABASE_URL;
       ```

       ```ts

         const app = express();

         const pool = new Pool({ connectionString, ssl: { rejectUnauthorized: false } });
       ```

    3. **Buat startServer() untuk local dev** — pindahkan semua startup logic:
       ```ts
       async function startServer() {
         const { app, pool } = createApp();
         await initDatabase(pool);
         await seedDefaultUser(pool);
         if (process.env.NODE_ENV !== "production") {
           const { createServer: createViteServer } = await import("vite");
           const vite = await createViteServer({ server: { middlewareMode: true }, appType: "spa" });
           app.use(vite.middlewares);
         } else {
           const distPath = path.join(process.cwd(), "dist");
           app.use(express.static(distPath));
           app.get("*", (req, res) => res.sendFile(path.join(distPath, "index.html")));
         }
         app.listen(PORT, "0.0.0.0", () => console.log(`Server fully running on http://localhost:${PORT}`));
       }
       ```
       Perhatikan: `createViteServer` import dipindah menjadi **dynamic import** di dalam if-block — hapus static import di top of file.

    4. **Update initDatabase dan seedDefaultUser signature** — tambahkan `pool` sebagai parameter karena pool sekarang dibuat di createApp():

<task type="auto">
  <name>Task 2: Buat api/index.ts dan vercel.json + update package.json</name>
  <files>api/index.ts, vercel.json, package.json</files>
  <action>
    **Buat `api/index.ts`** — Vercel serverless entry point:
    ```ts
    import path from "path";
    import { createApp } from "../server";

    let initialized = false;

    async function getInitializedApp() {
      const { app, pool } = createApp();
      if (!initialized) {
        const { initDatabase, seedDefaultUser } = await import("../server");
        await initDatabase(pool);
        await seedDefaultUser(pool);
        initialized = true;

        const distPath = path.join(process.cwd(), "dist");
        const express = await import("express");
        app.use(express.default.static(distPath));
        app.get("*", (_req, res) => res.sendFile(path.join(distPath, "index.html")));
      }
      return app;
    }

    export default async (req: any, res: any) => {
      const app = await getInitializedApp();
      app(req, res);
    };
    ```

    Catatan: initDatabase dan seedDefaultUser harus di-export dari server.ts (tambahkan `export` keyword pada kedua fungsi tersebut).

    **Buat `vercel.json`** di root project:
    ```json
    {
      "version": 2,
      "buildCommand": "npm run build",
      "outputDirectory": "dist",
      "functions": {
        "api/index.ts": {
          "maxDuration": 30
        }
      },
      "rewrites": [
        {
          "source": "/(.*)",
          "destination": "/api"
        }
      ]
    }
    ```
